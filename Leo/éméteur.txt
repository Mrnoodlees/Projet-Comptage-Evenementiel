#include <ESP8266WiFi.h>
#include <ESPAsyncWebServer.h>
#include <PubSubClient.h>
#include <EEPROM.h>
#include <ESP8266mDNS.h>

// --- CONFIGURATION MATÉRIELLE ---
const int PIN_IR = 4; // Pin D2 (GPIO 4)
const int PIN_BAT = A0;

struct Config {
  char ssid[32];
  char password[64];
  char deviceID[20];
  int frequence;  
  int timeON;   
  int timeOFF;  
  char mqttServer[100];
  int intervalBatSec;
  int intervalDiscoSec; 
  char webUser[20]; 
  char webPass[20]; 
  char staticIP[16];
  char gateway[16];
  char subnet[16];
  bool useStaticIP;
} settings;

WiFiClient espClient;
PubSubClient mqttClient(espClient);
AsyncWebServer server(80);

unsigned long lastBatMsg = 0;
unsigned long lastDiscoMsg = 0;
unsigned long lastBurst = 0; 
bool irState = false;          
bool portalMode = false;

// Conversion String vers IPAddress
IPAddress stringToIP(const char* s) {
  IPAddress ip;
  if (!ip.fromString(s)) return IPAddress(0,0,0,0);
  return ip;
}

void saveConfig() {
  EEPROM.begin(512);
  EEPROM.put(0, settings);
  EEPROM.write(500, 77); 
  EEPROM.commit();
  Serial.println("[EEPROM] Configuration sauvegardée !");
}

void loadConfig() {
  EEPROM.begin(512);
  EEPROM.get(0, settings);
  if (EEPROM.read(500) != 77 || settings.intervalBatSec <= 0) {
    Serial.println("[EEPROM] Initialisation par défaut...");
    strcpy(settings.ssid, "greta");
    strcpy(settings.password, "boulerouge62219!");
    strcpy(settings.mqttServer, "178.32.107.35");
    strcpy(settings.deviceID, "emetteur2");
    settings.frequence = 38;
    settings.timeON = 300;  
    settings.timeOFF = 1000; 
    settings.intervalBatSec = 10;
    settings.intervalDiscoSec = 60; 
    strcpy(settings.webUser, "admin");
    strcpy(settings.webPass, "admin123");
    strcpy(settings.staticIP, "192.168.1.100");
    strcpy(settings.gateway, "192.168.1.1");
    strcpy(settings.subnet, "255.255.255.0");
    settings.useStaticIP = false;
    saveConfig();
  }
}

void sendDiscovery() {
  if (mqttClient.connected()) {
    String mdnsURL = "http://" + String(settings.deviceID) + ".local";
    String payload = "{\"id\":\"" + String(settings.deviceID) + "\",\"url\":\"" + mdnsURL + "\",\"type\":\"emeteur\"}";
    mqttClient.publish("emeteur/discovery", payload.c_str(), true);
  }
}

void reconnectMQTT() {
  if (!mqttClient.connected()) {
    if (mqttClient.connect(settings.deviceID, "leo", "test")) {
      mqttClient.subscribe("porte/control"); 
      sendDiscovery();
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(500);
  loadConfig();

  pinMode(PIN_IR, OUTPUT);
  analogWrite(PIN_IR, 0); // Sécurité au démarrage
  analogWriteRange(1023); 
  analogWriteFreq(settings.frequence * 1000);
  
  WiFi.persistent(false);
  WiFi.disconnect();
  WiFi.mode(WIFI_STA);

  if(settings.useStaticIP) {
    WiFi.config(stringToIP(settings.staticIP), stringToIP(settings.gateway), stringToIP(settings.subnet));
  }

  WiFi.begin(settings.ssid, settings.password);

  int retry = 0;
  while (WiFi.status() != WL_CONNECTED && retry < 20) { delay(500); retry++; }

  if (WiFi.status() != WL_CONNECTED) {
    portalMode = true;
    WiFi.mode(WIFI_AP);
    WiFi.softAP("CONFIG_EMETTEUR", "admin123");
  } else {
    mqttClient.setServer(settings.mqttServer, 1883);
    if (MDNS.begin(settings.deviceID)) {
       MDNS.addService("http", "tcp", 80);
    }
  }

  // --- INTERFACE WEB ---
  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    if(!request->authenticate(settings.webUser, settings.webPass))
        return request->requestAuthentication();

    String h = "<html><head><meta charset='UTF-8'><style>";
    h += "body{font-family:sans-serif; background:#1a1a1a; color:white; text-align:center;}";
    h += "form{display:inline-block; text-align:left; background:#333; padding:20px; border-radius:10px; width:320px;}";
    h += "input{width:100%; margin-bottom:10px; padding:5px; border-radius:3px; border:none;}";
    h += "input[type=checkbox]{width:auto;}";
    h += "b{color:#00ff00; display:block; margin-top:10px;}";
    h += "hr{border:0; border-top:1px solid #555; margin:15px 0;}";
    h += "</style></head><body>";
    h += "<h2>Configuration " + String(settings.deviceID) + "</h2>";
    h += "<form action='/save' method='POST'>";
    
    h += "<b>Identité</b>";
    h += "ID:<input name='id' value='"+String(settings.deviceID)+"'>";

    h += "<hr><b>WiFi & MQTT</b>";
    h += "SSID:<input name='s' value='"+String(settings.ssid)+"'>";
    h += "Pass:<input type='password' name='p' value='"+String(settings.password)+"'>";
    h += "MQTT IP:<input name='m' value='"+String(settings.mqttServer)+"'>";
    h += "<span><input type='checkbox' name='uip' "+String(settings.useStaticIP?"checked":"")+"> Utiliser IP Fixe</span><br>";
    h += "IP Statique:<input name='ip' value='"+String(settings.staticIP)+"'>";
    h += "Passerelle:<input name='gw' value='"+String(settings.gateway)+"'>";
    h += "Masque:<input name='sn' value='"+String(settings.subnet)+"'>";

    h += "<hr><b>Sécurité Web</b>";
    h += "User:<input name='wu' value='"+String(settings.webUser)+"'>";
    h += "Pass:<input type='password' name='wp' value='"+String(settings.webPass)+"'>";

    h += "<hr><b>Signal IR</b>";
    h += "Fréq (kHz):<input type='number' name='f' value='"+String(settings.frequence)+"'>";
    h += "ON (us):<input type='number' name='ton' value='"+String(settings.timeON)+"'>";
    h += "OFF (us):<input type='number' name='toff' value='"+String(settings.timeOFF)+"'>";

    h += "<hr><b>Intervalles (sec)</b>";
    h += "Batterie:<input type='number' name='ibat' value='"+String(settings.intervalBatSec)+"'>";
    h += "Discovery:<input type='number' name='idisc' value='"+String(settings.intervalDiscoSec)+"'>";

    h += "<hr><input type='submit' value='SAUVEGARDER & REBOOT' style='background:green; color:white; font-weight:bold; cursor:pointer; padding:10px; width:100%; border-radius:5px;'>";
    h += "</form></body></html>";
    request->send(200, "text/html", h);
  });

  server.on("/save", HTTP_POST, [](AsyncWebServerRequest *request){
    if(!request->authenticate(settings.webUser, settings.webPass)) return request->requestAuthentication();
    
    if(request->hasParam("id", true)) strncpy(settings.deviceID, request->getParam("id", true)->value().c_str(), 19);
    if(request->hasParam("s", true)) strncpy(settings.ssid, request->getParam("s", true)->value().c_str(), 31);
    if(request->hasParam("p", true)) strncpy(settings.password, request->getParam("p", true)->value().c_str(), 63);
    if(request->hasParam("m", true)) strncpy(settings.mqttServer, request->getParam("m", true)->value().c_str(), 99);
    
    if(request->hasParam("ip", true)) strncpy(settings.staticIP, request->getParam("ip", true)->value().c_str(), 15);
    if(request->hasParam("gw", true)) strncpy(settings.gateway, request->getParam("gw", true)->value().c_str(), 15);
    if(request->hasParam("sn", true)) strncpy(settings.subnet, request->getParam("sn", true)->value().c_str(), 15);
    settings.useStaticIP = request->hasParam("uip", true);

    if(request->hasParam("wu", true)) strncpy(settings.webUser, request->getParam("wu", true)->value().c_str(), 19);
    if(request->hasParam("wp", true)) strncpy(settings.webPass, request->getParam("wp", true)->value().c_str(), 19);

    if(request->hasParam("f", true)) settings.frequence = request->getParam("f", true)->value().toInt();
    if(request->hasParam("ton", true)) settings.timeON = request->getParam("ton", true)->value().toInt();
    if(request->hasParam("toff", true)) settings.timeOFF = request->getParam("toff", true)->value().toInt();
    if(request->hasParam("ibat", true)) settings.intervalBatSec = request->getParam("ibat", true)->value().toInt();
    if(request->hasParam("idisc", true)) settings.intervalDiscoSec = request->getParam("idisc", true)->value().toInt();

    saveConfig();
    analogWrite(PIN_IR, 0); // Failsafe : on coupe l'IR avant reboot
    request->send(200, "text/plain", "Configuration OK. Reboot...");
    delay(2000); 
    ESP.restart();
  });

  server.begin();
  lastBurst = micros();
}

void loop() {
  yield(); 
  if (!portalMode) MDNS.update();

  // --- SIGNAL IR (Optimisé pour le comptage) ---
  unsigned long currentMicros = micros();
  unsigned long interval = irState ? (unsigned long)settings.timeON : (unsigned long)settings.timeOFF;
  
  if (currentMicros - lastBurst >= interval) {
    // On additionne pour éviter de cumuler les micro-retards d'exécution
    lastBurst += interval; 
    irState = !irState;
    analogWrite(PIN_IR, irState ? 512 : 0);
  }

  // --- MQTT & TIMERS ---
  if (!portalMode && WiFi.status() == WL_CONNECTED) {
    if (!mqttClient.connected()) {
      static unsigned long lastReconnect = 0;
      if (millis() - lastReconnect > 5000) {
        lastReconnect = millis();
        reconnectMQTT();
      }
    } else {
      mqttClient.loop();
      unsigned long now = millis();
      
      // Timer Batterie
      if (now - lastBatMsg > ((unsigned long)settings.intervalBatSec * 1000)) { 
        lastBatMsg = now;
        int vPourcent = constrain(map(analogRead(PIN_BAT), 450, 600, 0, 100), 0, 100); 
        String payload = "{\"id\":\"" + String(settings.deviceID) + "\",\"bat\":" + String(vPourcent) + "}";
        mqttClient.publish("emeteur/status", payload.c_str());
      }
      
      // Timer Discovery
      if (now - lastDiscoMsg > ((unsigned long)settings.intervalDiscoSec * 1000)) {
        lastDiscoMsg = now;
        sendDiscovery();
      }
    }
  }
}
